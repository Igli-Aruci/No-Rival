var output = {
    rewards: []
};
var skuTarget = "COP";

if (ikContext.account && ikContext.account.isDraft === true) {
    
    var targetPax = parseInt(ikContext.account.clientCount) || 0;
    var currentTotalQuantity = 0;
    var lines = ikContext.account.transactionLines || [];

    // Controllo più approfondito delle righe per trovare lo SKU (copre più varianti di API)
    for (var i = 0; i < lines.length; i++) {
        var line = lines[i];
        if (!line) continue;
        
        var currentSku = line.itemSku || line.sku || (line.item && line.item.sku) || (line.product && line.product.sku);
        
        if (currentSku === skuTarget) {
            currentTotalQuantity += (line.quantity || 0);
        }
    }

    // Se i coperti attuali sono DIVERSI da quelli desiderati
    if (targetPax !== currentTotalQuantity) {
        
        // Calcoliamo la differenza (sarà positiva se dobbiamo aggiungere, negativa se dobbiamo togliere)
        var delta = targetPax - currentTotalQuantity;
        
        // Generiamo il reward standard Teckel
        output.rewards.push({
            type: "ADD",
            sku: skuTarget,
            quantity: delta, // Lightspeed preferisce ricevere il delta esatto (es. +2 o -1)
            insertionPhase: 4
        });
    }
}

JSON.stringify(output);
