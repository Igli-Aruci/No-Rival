var output = {
    rewards: []
};
var sku = "COP";

// 1. Verifichiamo che il tavolo sia aperto (isDraft)
if (ikContext.account && ikContext.account.isDraft === true) {
    
    // Recuperiamo il numero di persone salvato nel tavolo
    var targetPax = ikContext.account.clientCount || 0;
    
    // 2. Calcoliamo quanti COP sono già presenti nel conto
    var currentTotalQuantity = 0;
    var lines = ikContext.account.transactionLines || [];

    for (var i = 0; i < lines.length; i++) {
        if (lines[i].itemSku === sku || lines[i].sku === sku) {
            currentTotalQuantity += (lines[i].quantity || 0);
        }
    }

    // 3. Se mancano dei coperti rispetto ai pax selezionati -> AGGIUNGIAMO
    if (targetPax > currentTotalQuantity) {
        var quantityToAdd = targetPax - currentTotalQuantity;
        
        output.rewards.push({
            type: "ADD",
            sku: sku,
            quantity: quantityToAdd,
            insertionPhase: 4
        });
    } 
    // 4. Se i coperti sono in ECCEDENZA -> Usiamo UPDATE per impostare il valore esatto dei PAX
    else if (currentTotalQuantity > targetPax) {
        
        output.rewards.push({
            type: "UPDATE", // Diciamo a Teckel di aggiornare lo stato di questo SKU
            sku: sku,
            quantity: targetPax, // Impostiamo direttamente il numero finale desiderato (es. 3)
            insertionPhase: 4
        });
    }
}

// Restituiamo il comando a Lightspeed
JSON.stringify(output);
