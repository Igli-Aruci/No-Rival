/* ===================================================================
   LIGHTSPEED GLOBAL TABLE MONITOR (ON_ACCOUNT_LOAD)
   =================================================================== */

var output = { rewards: [] };
var SKU_TARGET = "COP";

// Questo trigger si attiva all'ingresso del tavolo
if (typeof ikContext !== 'undefined' && ikContext && ikContext.account) {
    
    // Iniettiamo la sonda per leggere i PAX impostati sul display
    var targetPax = parseInt(ikContext.account.clientCount 
                    || (ikContext.table && ikContext.table.clientCount) 
                    || 0);

    var currentCoperti = 0;
    var lines = ikContext.account.transactionLines || [];

    // Scansione istantanea del carrello appena caricato a schermo
    for (var i = 0; i < lines.length; i++) {
        var line = lines[i];
        if (line && (line.voided !== true && line.deleted !== true)) {
            var activeSku = line.itemSku || line.sku || (line.item && line.item.sku);
            if (activeSku === SKU_TARGET) {
                currentCoperti += parseInt(line.quantity || 0);
            }
        }
    }

    // Sincronizzazione forzata del buffer
    var delta = targetPax - currentCoperti;

    if (delta !== 0) {
        output.rewards.push({
            type: "ADD", // Applica la variazione (es. +1 o -2)
            sku: SKU_TARGET,
            quantity: delta
        });
    }
}

// Spariamo il comando nel flusso di caricamento del tavolo
JSON.stringify(output);
