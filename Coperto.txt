var output = {
    rewards: []
};
var skuTarget = "COP";

if (ikContext.account && ikContext.account.isDraft === true) {
    
    var targetPax = parseInt(ikContext.account.clientCount) || 0;
    var lines = ikContext.account.transactionLines || [];
    
    var currentTotalQuantity = 0;
    var copertiLinesFound = [];

    // 1. Analizziamo il carrello SENZA MODIFICARLO (Sola lettura)
    for (var i = 0; i < lines.length; i++) {
        var line = lines[i];
        if (!line) continue;
        
        var currentSku = line.itemSku || line.sku || (line.item && line.item.sku);
        
        if (currentSku === skuTarget) {
            currentTotalQuantity += (line.quantity || 0);
            // Memorizziamo la riga originale per leggerne l'ID unico dopo
            copertiLinesFound.push(line); 
        }
    }

    // 2. CASO A: Mancano coperti -> AGGIUNGIAMO col metodo standard
    if (targetPax > currentTotalQuantity) {
        var quantityToAdd = targetPax - currentTotalQuantity;
        output.rewards.push({
            type: "ADD",
            sku: skuTarget,
            quantity: quantityToAdd,
            insertionPhase: 4
        });
    }
    
    // 3. CASO B: Troppi coperti -> RIMUOVIAMO puntando l'ID della riga specifica
    else if (currentTotalQuantity > targetPax) {
        var quantityToRemove = currentTotalQuantity - targetPax;
        
        // Scorriamo le righe dei coperti trovate per colpirle singolarmente tramite ID
        for (var j = copertiLinesFound.length - 1; j >= 0; j--) {
            var targetLine = copertiLinesFound[j];
            var lineQty = targetLine.quantity || 0;
            
            // Intercettiamo l'identificativo unico della riga generato da Lightspeed
            var exactLineId = targetLine.id || targetLine.lineId || targetLine.uuid;
            
            // Calcoliamo quanto togliere da questa specifica riga
            var toRemoveFromThisLine = Math.min(quantityToRemove, lineQty);
            
            output.rewards.push({
                type: "REMOVE",          // Comando di rimozione mirato
                sku: skuTarget,
                quantity: toRemoveFromThisLine,
                id: exactLineId,         // Diciamo al server l'ID esatto della riga da colpire
                lineId: exactLineId,     // Doppia sicurezza per compatibilità API
                insertionPhase: 4
            });
            
            quantityToRemove -= toRemoveFromThisLine;
            if (quantityToRemove === 0) break; // Abbiamo rimosso tutta l'eccedenza, stop
        }
    }
}

// Restituiamo il payload sicuro a Lightspeed
JSON.stringify(output);
