var output = {
    rewards: []
};
var skuTarget = "COP";

// Verifichiamo che il contesto esista e che l'account/tavolo sia modificabile (isDraft)
if (ikContext && ikContext.account && ikContext.account.isDraft === true) {
    
    // 1. FALLBACK DI SICUREZZA PER I PAX: Cerchiamo il numero di persone in tutte le posizioni possibili di Lightspeed
    var targetPax = ikContext.account.clientCount 
                    || (ikContext.table && ikContext.table.clientCount) 
                    || (ikContext.transaction && ikContext.transaction.clientCount) 
                    || 0;
                    
    var currentTotalQuantity = 0;
    var lines = ikContext.account.transactionLines || [];

    // 2. ISPEZIONE PROFONDA DEL CARRELLO: Estraiamo lo SKU in modo aggressivo
    for (var i = 0; i < lines.length; i++) {
        var line = lines[i];
        if (!line) continue;
        
        // Lightspeed K-Series mappa lo SKU in punti diversi a seconda della versione API
        var currentSku = line.itemSku 
                        || line.sku 
                        || (line.item && line.item.sku) 
                        || (line.product && line.product.sku)
                        || (line.catalogItem && line.catalogItem.sku);
        
        if (currentSku === skuTarget) {
            currentTotalQuantity += (line.quantity || 0);
        }
    }

    // 3. LOGICA DI SINCRO: Calcoliamo la differenza matematica (Delta)
    if (targetPax !== currentTotalQuantity) {
        var delta = targetPax - currentTotalQuantity;
        
        // Struttura formale Teckel Rich Item Reward
        var rewardAction = {
            type: "ADD", 
            sku: skuTarget,
            quantity: delta
        };

        // Rimuoviamo insertionPhase se bloccava l'esecuzione, lasciando che il motore decida in base all'endpoint rich-item
        output.rewards.push(rewardAction);
    }
}

// Ritorno della stringa JSON richiesta dall'interprete Lightspeed
JSON.stringify(output);
