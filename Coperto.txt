var output = {
    rewards: []
};
var SKU_TARGET = "COP";

// Verifichiamo l'esistenza del contesto e che il tavolo sia modificabile
if (typeof ikContext !== 'undefined' && ikContext && ikContext.account && ikContext.account.isDraft === true) {
    
    // Estraiamo il numero REALE di persone al tavolo (Buffer Diretto)
    var targetPax = parseInt(ikContext.account.clientCount 
                    || (ikContext.table && ikContext.table.clientCount) 
                    || 0);

    // Se ci sono persone al tavolo, ordiniamo a Lightspeed la quantità ESATTA.
    // Non facciamo somme, non guardiamo cosa c'è già nel carrello.
    if (targetPax > 0) {
        output.rewards.push({
            type: "ADD",
            sku: SKU_TARGET,
            quantity: targetPax, // Spariamo direttamente il numero finale (es. 3, o 4, o 2)
            insertionPhase: 4
        });
    }
}

// Iniezione del payload puro
JSON.stringify(output);
