var payload = {
    rewards: []
};
var TARGET_SKU = "COP";

if (typeof ikContext !== 'undefined' && ikContext && ikContext.account) {
    
    // Sniffing dei PAX reali aggiornati al millisecondo
    var realPax = parseInt(ikContext.account.clientCount || (ikContext.table && ikContext.table.clientCount) || 0);
    
    var currentStock = 0;
    var lines = ikContext.account.transactionLines || [];

    // Ispezione del carrello corrente
    for (var i = 0; i < lines.length; i++) {
        var node = lines[i];
        if (!node) continue;
        
        var activeSku = node.itemSku || node.sku || (node.item && node.item.sku) || (node.product && node.product.sku);
        if (activeSku === TARGET_SKU) {
            currentStock += (node.quantity || 0);
        }
    }

    // CALCOLO DELLA DEVIAZIONE
    var delta = realPax - currentStock;

    if (delta !== 0) {
        // Se i coperti sono maggiori del dovuto (es. stock=4, pax=3 -> delta=-1)
        // o se sono minori (es. stock=3, pax=4 -> delta=1)
        payload.rewards.push({
            type: "ADD", 
            sku: TARGET_SKU,
            quantity: delta // Iniettiamo il delta esatto (positivo o negativo)
        });
    }
}

// Iniezione del payload nel server
JSON.stringify(payload);
