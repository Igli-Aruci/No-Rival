var output = {
    rewards: []
};
var SKU_TARGET = "COP";

// Entriamo nel blocco di memoria protetto (ikContext)
if (typeof ikContext !== 'undefined' && ikContext && ikContext.account) {
    
    // Verifichiamo lo stato del draft (tavolo aperto e modificabile)
    if (ikContext.account.isDraft === true) {
        
        // Sonda multi-layer per intercettare i PAX reali aggiornati
        var targetPax = parseInt(ikContext.account.clientCount 
                        || (ikContext.table && ikContext.table.clientCount) 
                        || (ikContext.transaction && ikContext.transaction.clientCount) 
                        || 0);

        var currentValidCoperti = 0;
        var lines = ikContext.account.transactionLines || [];

        // SNIFFING PROFONDO DEL CARRELLO
        for (var i = 0; i < lines.length; i++) {
            var line = lines[i];
            if (!line) continue;

            // Estrazione dello SKU bypassando l'offuscamento delle proprietà di Lightspeed
            var activeSku = line.itemSku || line.sku || (line.item && line.item.sku) || (line.product && line.product.sku);
            
            if (activeSku === SKU_TARGET) {
                // BYPASS RIGHE FANTASMA: Escludiamo i coperti che sono stati stornati o cancellati dal sistema
                if (line.voided !== true && line.deleted !== true && line.isVoided !== true) {
                    currentValidCoperti += parseInt(line.quantity || 0);
                }
            }
        }

        // CALCOLO DELLA DEVIAZIONE DELLA MATRICE
        var delta = targetPax - currentValidCoperti;

        // Se c'è una qualsiasi asimmetria (es. PAX = 4, Coperti = 3 -> Delta = +1)
        // Oppure (es. PAX = 3, Coperti = 4 -> Delta = -1)
        if (delta !== 0) {
            output.rewards.push({
                type: "ADD", // Comando nativo accept-all
                sku: SKU_TARGET,
                quantity: delta, // Iniettiamo il delta esatto (positivo o negativo)
                insertionPhase: 4 // Forziamo l'aggancio alla fase di consolidamento dell'ordine
            });
        }
    }
}

// Spariamo il JSON crudo nel server Lightspeed
JSON.stringify(output);
