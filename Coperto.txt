    rewards: []
};

if (typeof ikContext !== 'undefined' && ikContext && ikContext.account) {
    
    var lines = ikContext.account.transactionLines || [];

    // Verifichiamo che ci sia la riga numero 2 (Indice 1)
    if (lines.length >= 2) {
        
        var targetLine = lines[1]; // Selezioniamo la seconda riga
        var exactLineId = targetLine.id || targetLine.lineId || targetLine.uuid;
        var skuTarget = targetLine.itemSku || targetLine.sku;

        // INIEZIONE DEL COMANDO CON MOTIVO DELLO STORNO
        output.rewards.push({
            type: "REMOVE",
            sku: skuTarget,
            quantity: targetLine.quantity || 1,
            id: exactLineId,
            lineId: exactLineId,
            
            // ─── BYPASS DEI MOTIVI DI STORNO (Proviamo i formati standard Lightspeed) ───

            reasonId: "ALTRO",                // ID ipotetico del motivo

                id: "ALTRO"
            },
            voidReasonId: "ALTRO",            // Campo flat per database cloud
            
            insertionPhase: 4
        });
    }
}
