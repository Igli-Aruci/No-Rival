// Verifichiamo che la memoria del carrello sia accessibile
if (typeof ikContext !== 'undefined' && ikContext && ikContext.account && ikContext.account.transactionLines) {
    
    // Sovrascriviamo l'array delle righe tenendo solo gli articoli che NON sono "COP"
    ikContext.account.transactionLines = ikContext.account.transactionLines.filter(function(line) {
        if (!line) return false;
        
        // Estraiamo lo SKU provando i vari layer di Lightspeed
        var currentSku = line.itemSku || line.sku || (line.item && line.item.sku) || (line.product && line.product.sku);
        
        // Se lo SKU è "COP", ritorna false e JavaScript lo ELIMINA dall'array
        return currentSku !== "COP";
    });
}

// Restituiamo una struttura vuota per i rewards per non mandare in errore l'endpoint
JSON.stringify({ rewards: [] });
